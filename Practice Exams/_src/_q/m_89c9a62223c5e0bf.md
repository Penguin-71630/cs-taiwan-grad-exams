```
template <class T> class ADTX
{
private:
  T *x;
  int f, r, cp;
public:
  ADTX(int C=10) { x = new T[C];  cp = C;  f = r = 0; }
  ~ADTX() { delete [] x; }
  bool isEmpty() const { return (f == r); }

};
```
