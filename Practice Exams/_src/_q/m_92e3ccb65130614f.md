int n=11;

int ptt(int x)
{
  int k=-1;
  int i=0;
  int j=n-1;
  do  {
    k=(i+j)/2;
    if (a[k]<=x)
      i=k+1;
    else
      j=k-1;
  } while (i<=j);
  return k;
}
